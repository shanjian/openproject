# OKR Board Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a per-project, activatable "OKR Board" page — a full work-package table (not Kanban) with Organization Unit and Version quick filters plus a 3-way hierarchy scope toggle — built as a new Rails engine module mirroring the Boards module's registration pattern, with an Angular page built on the same `PartitionedQuerySpacePageComponent`/`WorkPackageViewFiltersService` infrastructure the standard work-packages full view already uses.

**Architecture:** New Rails engine `modules/okr_board` registers a project module + `show_okr_board` permission + a single-action controller that resolves an availability gate server-side (exactly one `department`-format custom field enabled on an active type, plus at least one Version) before choosing to render a plain empty-state or the Angular bootstrap layout. The Angular side is a new lazy-loaded feature module with a 3-level ui-router state (isolated-query-space wrapper → page → list), reusing `WorkPackageListViewComponent` for the table and a new `OkrBoardFilterComponent` that computes the three hierarchy scopes from a client-side `parent_id → children` index and writes filter values directly via `WorkPackageViewFiltersService#replace`/`#remove` — the same generic mechanism that already reloads the table on any filter change, no board-specific filter service needed.

**Tech Stack:** Ruby on Rails (engine/module), RSpec, TypeScript/Angular (ui-router, RxJS), Jasmine/Karma, Capybara.

**Spec:** `docs/superpowers/specs/2026-08-17-okr-board-design.md`

## Global Constraints

- Organization Unit picker lists **top-level departments only** (`Group.organizational_units.where_detail(parent_id: nil)`).
- Exactly one `department`-format custom field, enabled on an active project type, must exist for the board to be usable — 0 or >1 is a configuration empty-state, never a silent "pick the first one".
- The three scopes are: `[unit.id]` (just this unit), `unit.self_and_ancestors.ids` (this and everything above — a no-op for a top-level unit today), `[unit.id] + unit.children.ids` (this and one level down, direct children only).
- Version options and the underlying `version_id` filter's accepted values must come from the same source — no separate, potentially-diverging scope.
- No Kanban board, no changes to the Boards module, no changes to the universal work-package filter sidebar.
- New permission `show_okr_board`, `dependencies: :view_work_packages` — granted to `default_role_non_member`, `default_role_member`, `default_role_reader` by default (mirrors `show_board_views`).
- Every step in this plan implicitly inherits these constraints.

```ruby
# modules/okr_board/app/models/okr_board/availability.rb
module OkrBoard
  class Availability
    def initialize(project)
      @project = project
    end

    def qualifying_custom_field
      candidates = @project
        .all_work_package_custom_fields
        .where(field_format: "department")
        .merge(WorkPackageCustomField.joins(:types).where(types: { id: @project.types }))
        .distinct

      candidates.count == 1 ? candidates.first : nil
    end

    def available?
      qualifying_custom_field.present? && @project.shared_versions.exists?
    end
  end
end
```

```ruby
# modules/okr_board/app/controllers/okr_board/pages_controller.rb
module OkrBoard
  class PagesController < ApplicationController
    before_action :find_project
    before_action :authorize
    before_action :authorize_work_package_permission, only: %i[index]

    def index
      if availability.available?
        render "index", layout: "angular/angular"
      else
        render "empty_state"
      end
    end

    private

    def find_project
      @project = Project.find(params[:project_id])
    end

    def availability
      @availability ||= OkrBoard::Availability.new(@project)
    end
    helper_method :availability

    def authorize_work_package_permission
      unless current_user.allowed_in_project?(:view_work_packages, @project)
        deny_access
      end
    end
  end
end
```

```typescript
// frontend/src/app/features/okr-board/okr-board-filter/okr-board-filter.component.ts
import { Component, OnInit } from '@angular/core';
import { firstValueFrom } from 'rxjs';
import {
  WorkPackageViewFiltersService,
} from 'core-app/features/work-packages/routing/wp-view-base/view-services/wp-view-filters.service';
import { ApiV3Service } from 'core-app/core/apiv3/api-v3.service';
import { ApiV3ResourceCollection } from 'core-app/core/apiv3/paths/apiv3-resource';
import { ApiV3Resource } from 'core-app/core/apiv3/cache/cachable-apiv3-resource';
import { HalResource } from 'core-app/features/hal/resources/hal-resource';
import { ApiV3FilterBuilder } from 'core-app/shared/helpers/api-v3/api-v3-filter-builder';
import { getPaginatedResults } from 'core-app/core/apiv3/helpers/get-paginated-results';

export const DEPARTMENT_FILTER_NAME = 'department';

@Component({
  selector: 'okr-board-filter',
  templateUrl: './okr-board-filter.component.html',
  standalone: false,
})
export class OkrBoardFilterComponent implements OnInit {
  topLevelUnits:HalResource[] = [];

  private allUnits:HalResource[] = [];

  private childrenIndex:Map<string, string[]> = new Map();

  constructor(
    readonly wpTableFilters:WorkPackageViewFiltersService,
    readonly apiV3Service:ApiV3Service,
  ) {
  }

  ngOnInit():void {
    void this.loadOrganizationalUnits();
  }

  childrenOf(unitId:string):string[] {
    return this.childrenIndex.get(unitId) || [];
  }

  private async loadOrganizationalUnits():Promise<void> {
    const filter = this.wpTableFilters.instantiate(DEPARTMENT_FILTER_NAME);
    const allowedValuesLink = (filter.schema.values!.allowedValues as { href:string }).href;

    this.allUnits = await firstValueFrom(
      getPaginatedResults<HalResource>(
        (params) => (this.apiV3Service.collectionFromString(allowedValuesLink) as
          ApiV3ResourceCollection<HalResource, ApiV3Resource>)
          .filtered(new ApiV3FilterBuilder(), params)
          .get(),
      ),
    );

    this.topLevelUnits = this.allUnits.filter((unit) => !unit.parent);
    this.childrenIndex = this.buildChildrenIndex(this.allUnits);
  }

  private buildChildrenIndex(units:HalResource[]):Map<string, string[]> {
    const index = new Map<string, string[]>();

    units.forEach((unit) => {
      const parentId = (unit.parent as HalResource | null)?.id as string | undefined;
      if (!parentId) {
        return;
      }

      const siblings = index.get(parentId) || [];
      siblings.push(unit.id as string);
      index.set(parentId, siblings);
    });

    return index;
  }
}
```

```typescript
export type OkrBoardScope = 'self' | 'ancestors' | 'children';

// inside the class:
selectedUnitId:string|null = null;

selectedScope:OkrBoardScope = 'self';

scopeValues(unitId:string, scope:OkrBoardScope):string[] {
  if (scope === 'children') {
    return [unitId, ...this.childrenOf(unitId)];
  }

  // 'self' and 'ancestors' are identical for a top-level-only picker —
  // there is nothing above a top-level unit to add.
  return [unitId];
}

onUnitChange(unitId:string|null):void {
  this.selectedUnitId = unitId;

  if (!unitId) {
    this.wpTableFilters.remove(DEPARTMENT_FILTER_NAME);
    return;
  }

  this.applyUnitFilter(unitId, this.selectedScope);
}

onScopeChange(scope:OkrBoardScope):void {
  this.selectedScope = scope;

  if (this.selectedUnitId) {
    this.applyUnitFilter(this.selectedUnitId, scope);
  }
}

private applyUnitFilter(unitId:string, scope:OkrBoardScope):void {
  const values = this.scopeValues(unitId, scope);

  this.wpTableFilters.replace(DEPARTMENT_FILTER_NAME, (filter) => {
    filter.operator = '=';
    filter.values = values;
  });
}
```

```typescript
// class field:
versions:HalResource[] = [];

// called from ngOnInit alongside loadOrganizationalUnits:
async loadVersions():Promise<void> {
  const filter = this.wpTableFilters.instantiate(VERSION_FILTER_NAME);
  const allowedValuesLink = (filter.schema.values!.allowedValues as { href:string }).href;

  this.versions = await firstValueFrom(
    getPaginatedResults<HalResource>(
      (params) => (this.apiV3Service.collectionFromString(allowedValuesLink) as
        ApiV3ResourceCollection<HalResource, ApiV3Resource>)
        .filtered(new ApiV3FilterBuilder(), params)
        .get(),
    ),
  );
}

onVersionChange(versionId:string|null):void {
  this.selectedVersionId = versionId;

  if (!versionId) {
    this.wpTableFilters.remove(VERSION_FILTER_NAME);
    return;
  }

  this.wpTableFilters.replace(VERSION_FILTER_NAME, (filter) => {
    filter.operator = '=';
    filter.values = [versionId];
  });
}
```

```typescript
syncSelectionFromLiveFilter():void {
  const filter = this.wpTableFilters.find(DEPARTMENT_FILTER_NAME);
  const currentValue = filter?.values?.[0] as string | undefined;

  if (!currentValue) {
    this.selectedUnitId = null;
    return;
  }

  const stillExists = this.allUnitIds().has(currentValue);
  if (stillExists) {
    this.selectedUnitId = currentValue;
    return;
  }

  // The filter references a unit that no longer exists in the loaded set
  // (e.g. its Group was deleted). Clear the live filter itself, not just
  // the displayed selection -- board-filter.component.ts's own
  // selectedQuickFilter only resets the display, leaving the table
  // filtered by a dead id. We deliberately don't repeat that here.
  this.selectedUnitId = null;
  this.wpTableFilters.remove(DEPARTMENT_FILTER_NAME);
}

private allUnitIds():Set<string> {
  return new Set(this.allUnits.map((unit) => unit.id as string));
}
```

